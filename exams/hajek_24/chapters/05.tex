\section{Termín 20. ledna 2025}

\subsection{Fourierova řada}
Nalezněte Fourierovu řadu pro periodické prodloužení funkce
\[
    f(t)=
    \begin{cases*}
        $t$, & $t \in \langle 0, \pi \rangle$ \\
        $0$, & $t \in \langle -\pi, 0 \rangle$
    \end{cases*}
\]
s periodou $2 \pi$.

\subsection{Stacionární body}
Nalezněte a vyšetřete kritické body funkce
\[
    f(x, y) = x^3 - x^2 + \frac{1}{2} y^2 - xy - 6x \text{.}
\]

\subsection{Křivkový integrál}
Vypočtěte
\[
    \int_{C} x e^{yz} \dif s \text{,}
\]
kde $C$ je úsečka z bodu $(0, 0, 0)$ do bodu $(1, 2, 3)$.

\subsection{Greenova věta}
Pomocí Greenovy věty vypočítejte křivkový integrál $\underset{C}{\int} \vec{F} \dif \vec{s}$, kde
\[
    \vec{F}(x,y) =
    \begin{pmatrix}
        -2xy^2 + \cos(x^2), & x^2 y
    \end{pmatrix}
\]
je vektorové pole a $C$ je kladně orientovaná hranice trojúhelníku s vrcholy $(0, 0), (2, 0)$ a $(1, 1)$. \\
Křivku $C$ načrtněte.

\subsection{Objem tělesa}
Najděte objem tělesa uvnitř válce $x^2 + y^2 = 1$, omezeného plochami $z=y^2$ a $z=0$.
